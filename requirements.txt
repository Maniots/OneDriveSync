# OneDrive PC Sync — Third-party dependencies
#
# pywin32 is required for the shutdown monitor (shutdown_monitor.py).
# The monitor uses win32gui, win32con, and win32api to create a hidden
# message-only window and run a Windows message loop; this is the
# Windows-recommended architecture for receiving WM_QUERYENDSESSION /
# WM_ENDSESSION notifications in interactive applications.  See:
#   https://learn.microsoft.com/en-us/windows/win32/shutdown/application-responsiveness
#
# Install it with: pip install pywin32
#
# All other dependencies (pathlib, dataclasses, typing, logging,
# subprocess, json, shutil, os, argparse, enum, msvcrt) are part of
# the Python 3.12+ standard library.
pywin32
